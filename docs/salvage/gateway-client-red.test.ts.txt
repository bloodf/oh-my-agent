// ─── Helpers ───────────────────────────────────────────────────────────────────

let _idSeq = 1;
function freshId(): number { return _idSeq++; }

/** Poll listAuthCredentials until predicate passes, timeout, or throws. */
async function waitFor(
  store: RemoteAuthCredentialStore,
  predicate: (creds: StoredAuthCredential[]) => boolean,
  timeoutMs = 2000,
  intervalMs = 50,
): Promise<void> {
  const deadline = Date.now() + timeoutMs;
  while (Date.now() < deadline) {
    const creds = store.listAuthCredentials();
    if (predicate(creds)) return;
    await new Promise((r) => setTimeout(r, intervalMs));
  }
  const creds = store.listAuthCredentials();
  if (!predicate(creds)) {
    throw new Error(
      `waitFor timed out after ${timeoutMs}ms; creds=${JSON.stringify(creds.map((c) => c.id))}`,
    );
  }
}

  // ── Credential scoping: listAuthCredentials filtered to bound ids ───────────

  test("RemoteAuthCredentialStore.listAuthCredentials sees only bound ids", async () => {
    const dir = await mkdtemp(join(tmpdir(), "gw-red-"));
    const dbPath = join(dir, "auth.db");
    const store = await SqliteAuthCredentialStore.open(dbPath);
    const storage = new AuthStorage(store);
    await storage.reload();

    const broker = startAuthBroker({
      storage,
      bearerTokens: ["admin-token"],
      version: "gw-test",
      disableRefresher: true,
    });

    const adminClient = new AuthBrokerClient({ url: broker.url, token: "admin-token" });
    const credA = await adminClient.uploadCredential("openai", { type: "api_key", key: "sk-a" });
    const credB = await adminClient.uploadCredential("anthropic", { type: "api_key", key: "sk-b" });
    const credC = await adminClient.uploadCredential("google", { type: "api_key", key: "sk-c" });

    const gw = await startCredentialGateway({
      upstreamUrl: broker.url,
      upstreamToken: "admin-token",
    });

    // Worker-1: bound to A and C, not B
    const w1Token = gw.issueWorkerToken({ workerId: "w1", credentialIds: [credA.id, credC.id] });

    const w1Client = new AuthBrokerClient({ url: gw.url, token: w1Token });
    const w1Store = new RemoteAuthCredentialStore({ client: w1Client, streamSnapshots: false });
    await w1Store.refreshSnapshot();

    try {
      const ids = w1Store.listAuthCredentials().map((c: StoredAuthCredential) => c.id);
      expect(ids).toContain(credA.id);
      expect(ids).toContain(credC.id);
      expect(ids).not.toContain(credB.id);
    } finally {
      w1Store.close();
      await gw.close();
      await broker.close();
      storage.close();
      await rm(dir, { recursive: true, force: true });
    }
  });

    const dir = await mkdtemp(join(tmpdir(), "gw-red-"));
    const dbPath = join(dir, "auth.db");
    const store = await SqliteAuthCredentialStore.open(dbPath);
    const storage = new AuthStorage(store);
    await storage.reload();

    const broker = startAuthBroker({
      storage,
      bearerTokens: ["admin-token"],
      version: "gw-test",
      disableRefresher: true,
    });

    const adminClient = new AuthBrokerClient({ url: broker.url, token: "admin-token" });
    const cred1 = await adminClient.uploadCredential("openai", { type: "api_key", key: "sk-s1" });
    const cred2 = await adminClient.uploadCredential("anthropic", { type: "api_key", key: "sk-s2" });
    const cred3 = await adminClient.uploadCredential("google", { type: "api_key", key: "sk-s3" });

    const gw = await startCredentialGateway({

    // Bound to cred1 and cred3 only

    const wClient = new AuthBrokerClient({ url: gw.url, token: wToken });
    const wStore = new RemoteAuthCredentialStore({
      client: wClient,
      streamSnapshots: true,
      backgroundIdleMs: 60_000, // keep stream alive for test
    });
    await wStore.refreshSnapshot(); // trigger initial snapshot load

    const ac = new AbortController();
    const events: SnapshotStreamSnapshotEvent[] = [];

    try {
      for await (const ev of wClient.openSnapshotStream({ signal: ac.signal })) {
        events.push(ev as SnapshotStreamSnapshotEvent);
        break; // first event only
      }

      expect(events).toHaveLength(1);
      const ids = events[0]!.credentials.map((c: SnapshotEntry) => c.id);
      expect(ids).toContain(cred1.id);
      expect(ids).toContain(cred3.id);
      expect(ids).not.toContain(cred2.id);
    } finally {
      ac.abort();
      wStore.close();
      await gw.close();
      await broker.close();
      storage.close();
      await rm(dir, { recursive: true, force: true });

    const dbPath = join(dir, "auth.db");

  // ── Shared disable: 409 + pending_policy + requester SSE full snapshot ───────

  test("shared credential disable returns 409 and requester store recovers credential", async () => {
    const dir = await mkdtemp(join(tmpdir(), "gw-red-"));
    const dbPath = join(dir, "auth.db");
    const store = await SqliteAuthCredentialStore.open(dbPath);
    const storage = new AuthStorage(store);
    await storage.reload();

    const broker = startAuthBroker({
      storage,
      bearerTokens: ["admin-token"],
      version: "gw-test",
      disableRefresher: true,
    });

    const adminClient = new AuthBrokerClient({ url: broker.url, token: "admin-token" });
    const sharedCred = await adminClient.uploadCredential("openai", { type: "api_key", key: "sk-shared" });

    const gw = await startCredentialGateway({
      upstreamUrl: broker.url,
      upstreamToken: "admin-token",
    });

    // Establish TWO active tokens both bound to sharedCred
    const w1Token = gw.issueWorkerToken({ workerId: "w1", credentialIds: [sharedCred.id] });
    const w2Token = gw.issueWorkerToken({ workerId: "w2", credentialIds: [sharedCred.id] });

    // Both workers get their own RemoteAuthCredentialStore with SSE enabled
    const w1Client = new AuthBrokerClient({ url: gw.url, token: w1Token });
    const w2Client = new AuthBrokerClient({ url: gw.url, token: w2Token });

    const w1Store = new RemoteAuthCredentialStore({
      client: w1Client,
      streamSnapshots: true,
      backgroundIdleMs: 60_000,
    });
    const w2Store = new RemoteAuthCredentialStore({
      client: w2Client,
      streamSnapshots: true,
      backgroundIdleMs: 60_000,
    });

    await w1Store.refreshSnapshot();
    await w2Store.refreshSnapshot();

    // Verify both see the credential before disable
    const idsBefore1 = w1Store.listAuthCredentials().map((c: StoredAuthCredential) => c.id);
    const idsBefore2 = w2Store.listAuthCredentials().map((c: StoredAuthCredential) => c.id);
    expect(idsBefore1).toContain(sharedCred.id);
    expect(idsBefore2).toContain(sharedCred.id);

    // Worker-1 calls deleteAuthCredential → triggers shared disable
    w1Store.deleteAuthCredential(sharedCred.id, "worker-shared-disable");

    // Wait for the requester's store to see the credential removed via SSE
    await waitFor(w1Store, (creds) => !creds.map((c) => c.id).includes(sharedCred.id));

    // Peer store must remain usable with the credential still visible
    const idsAfter2 = w2Store.listAuthCredentials().map((c: StoredAuthCredential) => c.id);
    expect(idsAfter2).toContain(sharedCred.id);

    // Worker-1's own store has already removed the credential (immediate local effect)
    const idsAfter1 = w1Store.listAuthCredentials().map((c: StoredAuthCredential) => c.id);
    expect(idsAfter1).not.toContain(sharedCred.id);

    w1Store.close();
    w2Store.close();
    await gw.close();
    await broker.close();
    storage.close();
    await rm(dir, { recursive: true, force: true });
  });

    const dir = await mkdtemp(join(tmpdir(), "gw-red-"));
    const dbPath = join(dir, "auth.db");
    const store = await SqliteAuthCredentialStore.open(dbPath);
    const storage = new AuthStorage(store);
    await storage.reload();

    const broker = startAuthBroker({
      storage,
      bearerTokens: ["admin-token"],
      version: "gw-test",
      disableRefresher: true,
    });

    const adminClient = new AuthBrokerClient({ url: broker.url, token: "admin-token" });
    const cred = await adminClient.uploadCredential("openai", { type: "api_key", key: "sk-approve" });

    const gw = await startCredentialGateway({
      upstreamUrl: broker.url,
      upstreamToken: "admin-token",
    });

    const wToken = gw.issueWorkerToken({ workerId: "w-appr", credentialIds: [cred.id] });

    const wClient = new AuthBrokerClient({ url: gw.url, token: wToken });
    const wStore = new RemoteAuthCredentialStore({
      client: wClient,
      streamSnapshots: false,
    });
    await wStore.refreshSnapshot();

    // Trigger pending disable
    wStore.deleteAuthCredential(cred.id, "pending-approve-test");

    const pending = gw.pendingDisables();
    expect(pending).toHaveLength(1);
    const { requestId } = pending[0]!;

    // Approve via daemon API
    await gw.approvePendingDisable(requestId);

    // Pending list cleared
    expect(gw.pendingDisables()).toHaveLength(0);

    // Upstream no longer has the credential
    const snap = await adminClient.fetchSnapshot();
    if (snap.status === 200) {
      const ids = snap.snapshot.credentials.map((c: SnapshotEntry) => c.id);
      expect(ids).not.toContain(cred.id);
    }

    wStore.close();
    await gw.close();
    await broker.close();
    storage.close();
    await rm(dir, { recursive: true, force: true });
  });
